import json, os, urllib.request
key=os.environ['SC_RL_BACKEND_API_KEY']

def post(path, payload):
    req=urllib.request.Request(
        'http://127.0.0.1:8093'+path,
        data=json.dumps(payload).encode('utf-8'),
        headers={'X-SC-RL-Key':key,'Content-Type':'application/json'},
        method='POST',
    )
    with urllib.request.urlopen(req,timeout=30) as r:
        return json.load(r)

for path in ('/v1/jobs/runtime','/v1/core/readiness'):
    req=urllib.request.Request('http://127.0.0.1:8093'+path,headers={'X-SC-RL-Key':key})
    with urllib.request.urlopen(req,timeout=30) as r:
        print(path)
        print(json.dumps(json.load(r),indent=2))

print('/v1/retrieval/plan')
print(json.dumps(post('/v1/retrieval/plan',{'query':'Compare solar storage versus wind storage','limit':5}),indent=2))
